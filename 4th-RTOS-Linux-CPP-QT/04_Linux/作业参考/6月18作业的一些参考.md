>1, 假设在Linux系统中, 在某个目录下有一个文件夹test, 这个test文件夹内部, 包含更多的子文件夹和子文件, 甚至子文件夹内部进一步包含更多的子文件夹和子文件,  统计在test中内部总共(子文件夹的文件夹也要统计)有多少文件夹(不统计. 和 ..) 
>
>```c
>int getDirNum(char * path){
>
>    int num = 0;
>    DIR *p = opendir(path);
>
>    while(1){
>        struct dirent *pDirent = readdir(p);
>        if(pDirent == NULL){
>            return num;
>        }
>        if(pDirent->d_type == DT_DIR 
>           && strcmp(pDirent->d_name, ".")!=0  && strcmp(pDirent->d_name, "..") != 0){
>            num++;
>
>            char buf[300] = {0};
>            sprintf(buf, "%s/%s", path, pDirent->d_name);
>            int sonSum = getDirNum(buf);
>
>            num += sonSum;
>        }
>    }
>    return 0;
>}
>
>int main(){
>    int sum = getDirNum("./test");
>    printf("sum: %d \n", sum);
>    return 0;
>}
>```

>2, 假设在Linux系统中, 在某个目录下有一个文件夹test, 这个test文件夹内部, 包含更多的子文件夹和子文件, 而且在不同的子文件夹的子文件夹中, 可能有多个都叫1.txt文件,  统计所有1.txt文件的内存总大小(所有1.txt文件大小加起来的和)
>
>```c
>int getSumNum(char * path){
>
>    int size = 0;
>    DIR *p = opendir(path);
>
>    while(1){
>        struct dirent *pDirent = readdir(p);
>        if(pDirent == NULL){
>            return size;
>        }
>        if(pDirent->d_type == DT_REG 
>           && strcmp(pDirent->d_name, "1.txt") == 0){
>
>            char buf[300] = {0};
>            sprintf(buf, "%s/%s", path, pDirent->d_name);
>
>            struct stat st;
>            stat(buf, &st);
>
>            size += st.st_size;
>
>        }else if(pDirent->d_type == DT_DIR 
>           && strcmp(pDirent->d_name, ".")!=0  && strcmp(pDirent->d_name, "..") != 0){
>
>            char buf[300] = {0};
>            sprintf(buf, "%s/%s", path, pDirent->d_name);
>            int sumSize = getSumNum(buf);
>            size += sumSize;
>            
>        }
>    }
>
>    return 0;
>}
>
>int main(){
>    int size = getSumNum("./test");
>    printf("size: %d \n", size);
>    return 0;
>}

# 1.txt

>```c
>// a.c
>#include <header.h>
>#include <sys/ioctl.h>
>
>int main(int argc, char *argv[argc]){
>
>    while(1){
>
>        int num = 0;
>        int file_fd = open("1.txt", O_RDWR);
>        read(file_fd, &num, sizeof(int));
>        if(num == 2){
>            // 读文件数据:打印
>            char buf[100] = {0};
>            read(file_fd, buf, sizeof(buf));
>            printf("%s \n", buf);
>
>            ftruncate(file_fd, 0);
>        }
>        close(file_fd);
>
>
>       int tag = 0;
>       ioctl(STDIN_FILENO, FIONREAD, &tag);
>       if(tag > 0){
>            // 读标准输入: 写文件
>            char buf[100] = {0};
>            read(STDIN_FILENO, buf, sizeof(buf));
>
>            int file_fd = open("1.txt", O_WRONLY);
>            num = 0;
>            write(file_fd, &num, sizeof(int));
>            write(file_fd, buf, sizeof(buf));
>
>            lseek(file_fd, 0, SEEK_SET);
>            num = 1;
>            write(file_fd, &num, sizeof(int));
>            close(file_fd);
>        }
>    }
>
>
>    return 0;
>}
>
>```

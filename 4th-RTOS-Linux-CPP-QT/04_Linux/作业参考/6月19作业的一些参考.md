# 1 比较文件一致

>编写C程序，判断任意两个文件，其内容是否是完全一致的。
>
>```c
>#include <header.h>
>
>int main(){
>
>    int file_fd1 = open("a.txt", O_RDWR);
>    int file_fd2 = open("b.txt", O_RDWR);
>
>    struct stat st1, st2;
>    stat("a.txt", &st1);
>    stat("b.txt", &st2);
>
>    if(st1.st_size  != st2.st_size){
>        printf("文件不一样1 \n");
>        return 0;
>
>    }
>
>    // 方式一
>    /*
>    while(1){
>        char buf1[100] = {0};
>        char buf2[100] = {0};
>
>        ssize_t ret1 = read(file_fd1, buf1, sizeof(buf1));
>        ssize_t ret2 = read(file_fd2, buf2, sizeof(buf2));
>        if(ret1 == 0 ){
>            break;
>        }
>        if(strncmp(buf1, buf2, ret1) != 0){
>            printf("文件不一样2 \n");
>            return 0;
>
>        }
>
>    }
>    */
>
>
>    // 方式二
>    while(1){
>        char buf1[100] = {0};
>        char buf2[100] = {0};
>
>        ssize_t ret1 = read(file_fd1, buf1, sizeof(buf1));
>        ssize_t ret2 = read(file_fd2, buf2, sizeof(buf2));
>        if(ret1 == 0 ){
>            break;
>        }
>        for(ssize_t i =0; i<ret1; i++){
>            if(buf1[i] != buf2[i]){
>                printf("文件不一样2 \n");
>                return 0;
>            }
>        }
>    }
>
>    printf("文件一样 \n");
>    close(file_fd1);
>    close(file_fd2);
>
>    return 0;
>}
>
>
>```

# 2 序列化

>定义一个学生结构体类型struct student，里边含有学号，姓名，分数；定义结构体数组struct student s[3],给数组赋初值后，写入文件。然后编写另一个程序打开文件，然后再读取文件进行打印输出.
>
>要求使用open, read, write来完成
>
>```c
>#include <header.h>
>typedef struct student_t{
>    int id;
>    float score;
>    char name[20];
>}student;
>
>int main(){
>
>    int fd = open("1.txt", O_RDWR|O_CREAT, 0600);
>
>    /*
>    // 写入方式一
>    student students[3] = {{1, 89.5, "zs"},{2,  90.5, "ls"},{3,  100, "zl"}};
>    write(fd, students, sizeof(students));
>
>    // 写入方式二
>    student students[3] = {{1, 89.5, "zs"},{2,  90.5, "ls"},{3,  100, "zl"}};
>    for(int i=0; i<3; i++){
>        write(fd, &students[i], sizeof(students[i]));
>    }
>
>    // 写入方式三
>    student students[3] = {{1, 89.5, "zs"},{2,  90.5, "ls"},{3,  100, "zl"}};
>    for(int i=0; i<3; i++){
>        write(fd, &students[i].id, sizeof(int));
>        write(fd, &students[i].score, sizeof(float));
>        write(fd, students[i].name, sizeof(students[i].name));
>    }
>    */
>
>// ------------------------------------------------------------------------
>
>    /*
>    // 读取方式一
>    student sts2[3];
>    read(fd, sts2, sizeof(sts2));
>    for(int i=0; i<3; i++){
>        printf("%d; %.2f; %s \n", sts2[i].id, sts2[i].score, sts2[i].name);
>    }
>
>    // 读取方式二
>    for(int i=0; i<3; i++){
>        student st;
>        read(fd, &st, sizeof(st));
>        printf("%d; %.2f; %s \n", st.id, st.score, st.name);
>    }
>    // 读取方式三
>    for(int i=0; i<3; i++){
>        int id = 0;
>        read(fd, &id, sizeof(int));
>        char buf[20]={0};
>        read(fd, buf, sizeof(buf));
>        float sc = 0;
>        read(fd, &sc, sizeof(float));
>        printf("id:%d; name:%s; score:%.2f \n", id, buf, sc);
>    }
>    */
>
>    close(fd);
>    return 0;
>}
>```

# 3 通信

>假设存在一个a程序和一个b程序, 同时启动这两个程序, 让这两个程序通过文件为媒介(比如1.txt, 比如2.txt为媒介, 一个媒介文件怎么实现?两个媒介文件怎么实现? )进行相互通信.
>
>做到:
>
>假设a程序有标准输入, 让a读取标准输入, 通过写文件, 可以让b读文件, 读到a写的内容打印.
>
>当b读到a传输过来的内容打印之后, 让b读取标准输入, 通过文件, 发送给a, a打印, 形成一问一答的通信方式
>(默认大家都不会输入过长的输入内容, 比如小于100字节)

### 3.1 只通过1.txt

>```c
>// a.c
>int main(){
>
>int file_fd = open("1.txt", O_WRONLY);
>int num = 1;
>write(file_fd, &num, sizeof(int));
>close(file_fd);
>
>while(1){
>
>  file_fd = open("1.txt", O_RDONLY);
>  num = 0;
>  read(file_fd, &num, sizeof(int));
>  close(file_fd);
>
>  if(num == 1){
>      // 读标准输入: 写文件
>      char buf[100] = {0};
>      read(STDIN_FILENO, buf, sizeof(buf));
>
>      int file_fd = open("1.txt", O_WRONLY);
>      num = 0;
>      write(file_fd, &num, sizeof(int));
>      write(file_fd, buf, sizeof(buf));
>      close(file_fd);
>      file_fd = open("1.txt", O_WRONLY);
>      num = 2;
>      write(file_fd, &num, sizeof(int));
>      close(file_fd);
>  }else if(num == 4){
>      // 读文件数据:打印
>      file_fd = open("1.txt", O_RDONLY);
>      num = 0;
>      read(file_fd, &num, sizeof(int));
>      char buf[100] = {0};
>      read(file_fd, buf, sizeof(buf));
>      printf("%s \n", buf);
>      close(file_fd);
>
>      int file_fd = open("1.txt", O_WRONLY);
>      num = 1;
>      write(file_fd, &num, sizeof(int));
>       close(file_fd);
>  }
>}
>
>return 0;
>}
>```

### 3.2 通过1.txt 和 2.txt

>```c
>// a.c
>void my_read(int file_fd, char *buf, ssize_t len){
>
>   ssize_t size = 0;
>   while(size < len){
>       ssize_t ret = read(file_fd, buf+size, len-size);
>
>       size += ret;
>   }
>}
>int main(){
>
>
>while(1){
>
>   // 读标准输入: 写文件
>   char buf[100] = {0};
>   read(STDIN_FILENO, buf, sizeof(buf));
>
>   int file1 = open("1.txt", O_WRONLY);
>   write(file1, buf, sizeof(buf));
>   close(file1);
>
>
>   // 读文件打印
>   int file2 = open("2.txt", O_RDWR);
>   bzero(buf, sizeof(buf));
>   my_read(file2, buf, sizeof(buf));
>   ftruncate(file2, 0);
>   close(file2);
>   printf("%s \n", buf);
>
>}
>
>return 0;
>}
